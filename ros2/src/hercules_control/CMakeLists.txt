cmake_minimum_required(VERSION 3.16)
project(hercules_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(airsim_interfaces REQUIRED)

# Deliberately no ament_target_dependencies on the core target.
add_library(hercules_control_core src/core.cpp)
target_include_directories(hercules_control_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(hercules_control_core PUBLIC Eigen3::Eigen)
add_library(hercules_control_adapter src/ros_adapter.cpp)
target_link_libraries(hercules_control_adapter PUBLIC hercules_control_core)
ament_target_dependencies(hercules_control_adapter PUBLIC nav_msgs airsim_interfaces)
add_executable(smoke_node src/smoke_node.cpp)
target_link_libraries(smoke_node PUBLIC hercules_control_adapter)
ament_target_dependencies(smoke_node PUBLIC rclcpp)
foreach(target hercules_control_core hercules_control_adapter smoke_node)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
install(TARGETS hercules_control_core hercules_control_adapter EXPORT hercules_controlTargets
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(TARGETS smoke_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_export_targets(hercules_controlTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(Eigen3 nav_msgs airsim_interfaces)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_core test/test_core.cpp)
  target_link_libraries(test_core hercules_control_core)
  ament_add_gtest(test_adapter test/test_adapter.cpp)
  target_link_libraries(test_adapter hercules_control_adapter)
  ament_add_pytest_test(test_transport test/test_transport.py TIMEOUT 120
    ENV "SMOKE_EXE=$<TARGET_FILE:smoke_node>" "ROS_DOMAIN_ID=143" "ROS_LOCALHOST_ONLY=1")
endif()
ament_package()
