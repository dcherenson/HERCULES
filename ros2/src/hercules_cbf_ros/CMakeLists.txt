cmake_minimum_required(VERSION 3.16)
project(hercules_cbf_ros VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(hercules_cbf CONFIG REQUIRED)
find_package(hercules_interfaces REQUIRED)
find_package(std_msgs REQUIRED)

add_library(${PROJECT_NAME} src/cbf_adapter.cpp src/obstacle_cache.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(${PROJECT_NAME} PUBLIC hercules_cbf::hercules_cbf Eigen3::Eigen)
ament_target_dependencies(${PROJECT_NAME} PUBLIC builtin_interfaces hercules_interfaces std_msgs)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS ${PROJECT_NAME} EXPORT ${PROJECT_NAME}Targets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(PROGRAMS scripts/obstacle_observer_node.py
  DESTINATION lib/${PROJECT_NAME} RENAME obstacle_observer_node)
install(PROGRAMS scripts/mission_collision_observer_node.py
  DESTINATION lib/${PROJECT_NAME} RENAME mission_collision_observer_node)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_targets(${PROJECT_NAME}Targets HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Eigen3 hercules_cbf hercules_interfaces)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_cbf_adapter test/test_cbf_adapter.cpp)
  target_link_libraries(test_cbf_adapter ${PROJECT_NAME})
  ament_add_gtest(test_obstacle_cache test/test_obstacle_cache.cpp)
  target_link_libraries(test_obstacle_cache ${PROJECT_NAME})
  ament_add_pytest_test(test_obstacle_observer test/test_obstacle_observer.py TIMEOUT 60)
  ament_add_pytest_test(test_obstacle_observer_worker test/test_obstacle_observer_worker.py TIMEOUT 60)
  ament_add_pytest_test(test_obstacle_observer_transport test/test_obstacle_observer_transport.py TIMEOUT 60)
  ament_add_pytest_test(test_collision_observer test/test_collision_observer.py TIMEOUT 60)
  ament_add_pytest_test(test_fleet_configuration test/test_fleet_configuration.py TIMEOUT 60)
endif()
ament_package()
